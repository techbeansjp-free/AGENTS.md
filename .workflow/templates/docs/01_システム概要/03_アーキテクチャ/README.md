    User((ユーザー))

    subgraph DevEnv["開発環境"]
        subgraph Frontend["フロントエンド"]
            BFF["BFF / Web アプリ"]
        end
        subgraph Backend["バックエンド"]
            API["API サーバー"]
            Worker["ワーカー（任意）"]
        end
        subgraph Infra["インフラ"]
            DB[("データベース")]
            Queue["キュー（任意）"]
        end
    end

    User -->|"HTTPS"| BFF
    BFF -->|"API 呼び出し"| API
    API --> DB
    Queue -.->|"タスク供給"| Worker
